Dependencies and validation
==================

blueprint {
  version = "1.0.0"
  name = "rust-binary"
  description = "Rust binary project template"
}

prompt "project_name" {
  type = string
  help = "Name of your Rust project"
  default = "my-project"

  validate {
    regex = "^[a-z][a-z0-9_-]*$"
    min_length = 3
    max_length = 50
  }
}

prompt "features" {
  type = multiselect
  help = "Select features to include"
  choices = ["logging", "async", "cli", "config",]

  validate {
    min_selections = 1
    max_selections = 3
  }
}

prompt "cli_framework" {
  type = select
  help = "Choose CLI framework"
  choices = ["clap", "structopt", "argh"]
  default = "clap"
  depends_on = all(
    features.contains("cli"),
    any(
      project_name != "example",
      project_name == "demo",
    ),
  )
}
